// Independent integer-coordinate oracles: production scaling/centering helpers are never used.
let state = 19;
function random(): number { state = (Math.imul(state, 1664525) + 1013904223) >>> 0; return state; }
function mean(values: number[]): number { return values.reduce((total, value) => total + value, 0) / values.length; }
function pairwiseVariance(values: number[]): number {
  let total = 0;
  for (let left = 0; left < values.length; left++) for (let right = left + 1; right < values.length; right++) total += (values[left]! - values[right]!) ** 2;
  return total / values.length ** 2;
}
function directCenteredCorrelation(x: number[], y: number[]): number | null {
  const mx = mean(x), my = mean(y);
  let numerator = 0, ex = 0, ey = 0;
  for (let index = 0; index < x.length; index++) {
    const left = x[index]! - mx, right = y[index]! - my;
    numerator += left * right; ex += left * left; ey += right * right;
  }
  return ex && ey ? numerator / Math.sqrt(ex * ey) : null;
}
let caseCount = 0, highOffsetStatisticsCaseCount = 0, maxVarianceRelativeError = 0, maxCoefficientError = 0, undefinedMismatchCount = 0;
for (let trial = 0; trial < 1000; trial++) {
  const count = 2 + random() % 18, a = Array.from({ length: count }, () => random() % 41 - 20), b = Array.from({ length: count }, () => random() % 41 - 20);
  const times = a.map((_, index) => index * .25);
  for (const variant of ['high-offset', 'subnormal'] as const) {
    const unit = variant === 'high-offset' ? 2 ** -13 : Number.MIN_VALUE, offset = variant === 'high-offset' ? 9e11 : 0;
    const x = a.map(value => offset + value * unit), y = b.map(value => -offset + value * unit);
    const report = analyzeSignalCorrelation({ times, x, y }, { maxLag: 0, minOverlap: 2, removeMean: true }), expected = directCenteredCorrelation(a, b), actual = report.rows[0]!.coefficient;
    if ((expected === null) !== (actual === null)) undefinedMismatchCount++;
    if (expected !== null && actual !== null) maxCoefficientError = Math.max(maxCoefficientError, Math.abs(expected - actual));
    if (variant === 'high-offset') {
      const statistics = analyzeTimeStatistics({ times, values: x }), expectedVariance = pairwiseVariance(a) * unit * unit;
      if (expectedVariance) maxVarianceRelativeError = Math.max(maxVarianceRelativeError, Math.abs(statistics.variancePopulation - expectedVariance) / expectedVariance);
      else assert.equal(statistics.variancePopulation, 0);
      highOffsetStatisticsCaseCount++;
    }
    caseCount++;
  }
}
